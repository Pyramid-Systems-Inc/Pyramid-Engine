function(add_image_test target_name test_name source_file)
    add_executable(${target_name} ${source_file})
    target_link_libraries(${target_name} PRIVATE Pyramid::Image)
    target_include_directories(${target_name} PRIVATE "${PROJECT_SOURCE_DIR}/Tests")
    target_compile_definitions(${target_name} PRIVATE PYRAMID_TEST_SOURCE_DIR="${PROJECT_SOURCE_DIR}")
    set_target_properties(${target_name} PROPERTIES FOLDER "Tests/Image")
    add_test(NAME Image.${test_name} COMMAND $<TARGET_FILE:${target_name}>)
endfunction()
